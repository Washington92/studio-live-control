cmake_minimum_required(VERSION 3.20)
project(WMixEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(wmix-engine src/main.cpp)
if(MSVC)
  target_compile_options(wmix-engine PRIVATE /W4)
else()
  target_compile_options(wmix-engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME engine_selftest COMMAND wmix-engine --selftest)

option(WMIX_WITH_LIBOBS "Build experimental libobs smoke test" OFF)
if(WMIX_WITH_LIBOBS)
  add_subdirectory(obs)
endif()
